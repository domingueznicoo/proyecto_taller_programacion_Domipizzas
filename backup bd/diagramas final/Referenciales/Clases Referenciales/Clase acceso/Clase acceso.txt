class "GUI Login" <<boundary>> {
  +txt_username : String
  +txt_password : String
  +btn_acceder() : void
}

class AccesoController <<control>> {
  +validarCampos() : boolean
  +validarCredenciales() : boolean
  +verificarUsuarioActivo() : boolean
  +verificarSucursalActiva() : boolean
  +crearSesion() : void
  +registrarBitacora() : void
}

class conexion <<control>> {
  -host : String
  -puerto : int
  -nombre_bd : String
  -contraseña : String
  +conector() : Connection
}

class usuarios <<entity>> {
  +id_usuario : int
  +username : String
  +usua_password : String
  +estado_usuario : String
  +id_personal : int
  +id_sucursal : int
  +modulo_id : int
}

class personal <<entity>> {
  +id_personal : int
  +personal_nombre : String
  +personal_apellido : String
  +personal_ci : String
  +personal_telefono : String
  +personal_estado : String
}

class sucursales <<entity>> {
  +id_sucursal : int
  +descripcion_sucursal : String
  +estado_sucursal : String
}

class modulos <<entity>> {
  +modulo_id : int
  +modulo_descri : String
}

class bitacora <<entity>> {
  +id_bitacora : int
  +id_usuario : int
  +accion_realizada : String
  +fecha_accion : DateTime
  +ip_origen : String
  +navegador_usuario : String
}
