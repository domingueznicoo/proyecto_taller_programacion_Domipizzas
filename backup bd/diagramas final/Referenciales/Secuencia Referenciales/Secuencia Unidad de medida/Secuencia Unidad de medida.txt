@startuml
title Mantener Unidad de Medida – Botón: Guardar / Editar / Eliminar

autonumber

actor "Administrador" as ADM
boundary "GUI Lista Unidad de Medida" as GUILista
boundary "GUI Unidad de Medida" as GUIForm
control "UnidadMedidaController" as CTRL
control "conexion" as CONX
entity "unidad_medida" as UM
entity "usuarios" as USU
entity "bitacora" as BIT

ADM -> GUILista : Accede a Referenciales → Compras → Unidad de medida
GUILista -> CTRL : listarUnidades()

CTRL -> CONX : conector()
CONX --> CTRL : Connection

CTRL -> UM : consultar_unidades()
UM --> CTRL : lista_unidades

CTRL -> GUILista : mostrar lista principal

ADM -> GUILista : Clic en "Agregar Unidad de Medida"
GUILista -> GUIForm : abrir formulario

GUIForm -> CTRL : generarCodigo()
CTRL -> UM : obtener_ultimo_id_unidad()
UM --> CTRL : ultimo_id
CTRL -> GUIForm : mostrar código correlativo

ADM -> GUIForm : Ingresa descripción
ADM -> GUIForm : Clic en "Guardar"

GUIForm -> CTRL : validarDescripcion(unidad_descri)
CTRL --> GUIForm : descripción válida / inválida
GUIForm -> ADM : Mostrar error si descripción vacía

GUIForm -> CTRL : registrarUnidad(unidad_descri)

CTRL -> USU : consultar_usuario(id_usuario)
USU --> CTRL : datos_usuario

CTRL -> UM : insertar_unidad(unidad_descri, id_usuario)
UM --> CTRL : registro exitoso

CTRL -> BIT : registrarBitacora(id_usuario, accion_realizada, fecha_accion)
BIT --> CTRL : registro exitoso

CTRL -> GUIForm : mostrar confirmación
GUIForm -> GUILista : actualizar lista principal

ADM -> GUILista : Selecciona unidad y clic en "Editar"
GUILista -> CTRL : obtenerUnidad(id_unidad)

CTRL -> UM : consultar_unidad(id_unidad)
UM --> CTRL : datos_unidad

CTRL -> GUIForm : mostrar formulario con datos cargados

ADM -> GUIForm : Modifica descripción
ADM -> GUIForm : Clic en "Guardar"

GUIForm -> CTRL : validarDescripcion(unidad_descri)
CTRL --> GUIForm : descripción válida / inválida
GUIForm -> ADM : Mostrar error si descripción vacía

GUIForm -> CTRL : editarUnidad(id_unidad, unidad_descri)

CTRL -> UM : actualizar_unidad(id_unidad, unidad_descri)
UM --> CTRL : actualización exitosa

CTRL -> BIT : registrarBitacora(id_usuario, accion_realizada, fecha_accion)
BIT --> CTRL : registro exitoso

CTRL -> GUIForm : mostrar confirmación
GUIForm -> GUILista : actualizar lista principal

ADM -> GUILista : Selecciona unidad y clic en "Eliminar"
GUILista -> ADM : Solicitar confirmación

ADM -> GUILista : Confirma eliminación
GUILista -> CTRL : eliminarUnidad(id_unidad)

CTRL -> UM : eliminar_unidad(id_unidad)
UM --> CTRL : eliminación exitosa

CTRL -> BIT : registrarBitacora(id_usuario, accion_realizada, fecha_accion)
BIT --> CTRL : registro exitoso

CTRL -> GUILista : mostrar confirmación o error
GUILista -> ADM : actualizar lista principal

@enduml
