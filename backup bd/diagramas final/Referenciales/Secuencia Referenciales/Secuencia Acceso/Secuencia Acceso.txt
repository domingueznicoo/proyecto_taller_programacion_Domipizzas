@startuml
title Acceso – Botón: Acceder

autonumber

actor "Usuario del sistema" as USR
boundary "GUI Login" as GUI
control "AccesoController" as CTRL
control "conexion" as CONX
entity "usuarios" as USU
entity "personal" as PER
entity "sucursales" as SUC
entity "modulos" as MOD
entity "bitacora" as BIT

USR -> GUI : Inicia acceso al sistema
GUI -> GUI : Muestra formulario de login

USR -> GUI : Ingresa nombre de usuario
USR -> GUI : Ingresa contraseña
USR -> GUI : Clic en "Acceder"

GUI -> CTRL : validarCampos(username, password)
CTRL --> GUI : campos válidos / inválidos
GUI -> USR : Mostrar mensaje de error si existen campos vacíos

GUI -> CTRL : validarCredenciales(username, password)
CTRL -> CONX : conector()
CONX --> CTRL : Connection

CTRL -> USU : consultar_por_username(username)
USU --> CTRL : id_usuario, username, usua_password, estado_usuario, id_personal, id_sucursal, modulo_id

CTRL --> GUI : Informar credenciales incorrectas si usuario no existe
GUI -> USR : Mostrar credenciales incorrectas

CTRL -> CTRL : validarCredenciales(password, usua_password)
CTRL --> GUI : Informar credenciales incorrectas si contraseña incorrecta
GUI -> USR : Mostrar credenciales incorrectas

CTRL -> CTRL : verificarUsuarioActivo(estado_usuario)
CTRL --> GUI : Informar acceso bloqueado si usuario inactivo
GUI -> USR : Mostrar usuario inactivo

CTRL -> PER : consultar_personal(id_personal)
PER --> CTRL : id_personal, personal_estado

CTRL --> GUI : Informar acceso bloqueado si perfil inactivo
GUI -> USR : Mostrar perfil inactivo

CTRL -> SUC : consultar_sucursal(id_sucursal)
SUC --> CTRL : id_sucursal, descripcion_sucursal, estado_sucursal

CTRL -> CTRL : verificarSucursalActiva(estado_sucursal)
CTRL --> GUI : Informar acceso bloqueado si sucursal inactiva
GUI -> USR : Mostrar sucursal inactiva

CTRL -> MOD : consultar_modulo(modulo_id)
MOD --> CTRL : modulo_id, modulo_descri

CTRL --> GUI : Informar acceso bloqueado si módulo no asignado
GUI -> USR : Mostrar módulo no asignado

CTRL -> BIT : registrarBitacora(id_usuario, fecha_accion, id_sucursal, modulo_id)
BIT --> CTRL : registro exitoso

CTRL -> GUI : crearSesion(usuario, personal, sucursal, modulo)
GUI -> USR : Redirigir a Menú Principal

@enduml
