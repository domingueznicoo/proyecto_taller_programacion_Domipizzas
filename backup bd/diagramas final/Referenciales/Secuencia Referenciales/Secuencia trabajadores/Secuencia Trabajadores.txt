@startuml
title Mantener Trabajadores – Botón: Guardar / Editar / Eliminar

autonumber

actor "Administrador del sistema" as ADM
boundary "GUI Lista de Trabajadores" as GUILista
boundary "GUI Trabajadores" as GUIForm
control "TrabajadorController" as CTRL
control "conexion" as CONX
entity "trabajadores" as TRAB
entity "personal" as PER
entity "etapa_produccion" as ETAPA
entity "bitacora" as BIT

ADM -> GUILista : Accede a Producción → Referenciales → Trabajadores
GUILista -> CTRL : listarTrabajadores()
CTRL -> CONX : conector()
CONX --> CTRL : Connection
CTRL -> TRAB : consultar_trabajadores()
TRAB --> CTRL : lista_trabajadores
CTRL -> GUILista : mostrar lista principal

ADM -> GUILista : Clic en "Nuevo"
GUILista -> GUIForm : abrir formulario de registro

GUIForm -> CTRL : cargarReferenciales()
CTRL -> PER : consultar_personal_activo()
PER --> CTRL : lista_personal
CTRL -> ETAPA : consultar_etapas_produccion()
ETAPA --> CTRL : lista_etapas
CTRL -> GUIForm : mostrar referenciales

ADM -> GUIForm : Ingresa personal, rol, turno, costo hora y etapa
GUIForm -> CTRL : validarCampos(datos_trabajador)
CTRL --> GUIForm : campos válidos / inválidos
GUIForm -> ADM : Mostrar error si existen campos inválidos

ADM -> GUIForm : Clic en "Guardar"
GUIForm -> CTRL : registrarTrabajador(datos_trabajador)

CTRL -> PER : verificar_personal(id_personal)
PER --> CTRL : datos_personal

CTRL -> ETAPA : verificar_etapa(id_etapa)
ETAPA --> CTRL : datos_etapa

CTRL -> TRAB : insertar_trabajador(id_personal, trabajador_rol, trabajador_turno, trabajador_costo_hora, id_etapa)
TRAB --> CTRL : trabajadores_id

CTRL -> BIT : registrarBitacora(id_usuario, accion_realizada, fecha_accion)
BIT --> CTRL : registro exitoso

CTRL -> GUIForm : mostrar confirmación
GUIForm -> GUILista : actualizar lista principal

ADM -> GUILista : Selecciona trabajador y clic en "Editar"
GUILista -> CTRL : obtenerTrabajador(trabajadores_id)
CTRL -> TRAB : consultar_trabajador(trabajadores_id)
TRAB --> CTRL : datos_trabajador
CTRL -> GUIForm : mostrar formulario con datos cargados

ADM -> GUIForm : Modifica datos y clic en "Guardar"
GUIForm -> CTRL : editarTrabajador(trabajadores_id, datos_trabajador)

CTRL -> PER : verificar_personal(id_personal)
PER --> CTRL : datos_personal

CTRL -> ETAPA : verificar_etapa(id_etapa)
ETAPA --> CTRL : datos_etapa

CTRL -> TRAB : actualizar_trabajador(trabajadores_id, datos_trabajador)
TRAB --> CTRL : actualización exitosa

CTRL -> BIT : registrarBitacora(id_usuario, accion_realizada, fecha_accion)
BIT --> CTRL : registro exitoso

CTRL -> GUIForm : mostrar confirmación
GUIForm -> GUILista : actualizar lista principal

ADM -> GUILista : Selecciona trabajador y clic en "Eliminar"
GUILista -> ADM : Solicitar confirmación

ADM -> GUILista : Confirma eliminación
GUILista -> CTRL : eliminarTrabajador(trabajadores_id)

CTRL -> TRAB : validar_vinculos(trabajadores_id)
TRAB --> CTRL : sin vínculos / con vínculos

CTRL -> TRAB : eliminar_o_inactivar_trabajador(trabajadores_id)
TRAB --> CTRL : eliminación/inactivación exitosa

CTRL -> BIT : registrarBitacora(id_usuario, accion_realizada, fecha_accion)
BIT --> CTRL : registro exitoso

CTRL -> GUILista : mostrar confirmación o error
GUILista -> ADM : actualizar lista principal

@enduml
