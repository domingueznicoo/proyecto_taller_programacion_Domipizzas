@startuml
title Mantener Usuarios – Botón: Guardar / Editar / Eliminar

autonumber

actor "Administrador del sistema" as ADM
boundary "GUI Lista de Usuarios" as GUILista
boundary "GUI Usuarios" as GUIForm
control "UsuarioController" as CTRL
control "conexion" as CONX
entity "usuarios" as USU
entity "personal" as PER
entity "sucursales" as SUC
entity "cargos" as CAR
entity "modulos" as MOD
entity "bitacora" as BIT

ADM -> GUILista : Accede al mantenimiento de usuarios
GUILista -> CTRL : listarUsuarios()
CTRL -> CONX : conector()
CONX --> CTRL : Connection
CTRL -> USU : consultar_usuarios()
USU --> CTRL : lista_usuarios
CTRL -> GUILista : mostrar lista de usuarios

ADM -> GUILista : Clic en "Nuevo Usuario"
GUILista -> GUIForm : abrir formulario de registro

ADM -> GUIForm : Ingresa personal, username, password, confirmación, sucursal, cargo y módulo
GUIForm -> CTRL : validarCampos(datos_usuario)
CTRL --> GUIForm : campos válidos / inválidos
GUIForm -> ADM : Mostrar error si existen campos inválidos

ADM -> GUIForm : Clic en "Guardar"
GUIForm -> CTRL : registrarUsuario(datos_usuario)

CTRL -> PER : verificar_personal(id_personal)
PER --> CTRL : datos_personal

CTRL -> SUC : verificar_sucursal(id_sucursal)
SUC --> CTRL : descripcion_sucursal, estado_sucursal

CTRL -> CAR : verificar_cargo(id_cargo)
CAR --> CTRL : cargo_descripcion, estado_cargo

CTRL -> MOD : verificar_modulo(modulo_id)
MOD --> CTRL : modulo_id, modulo_descri

CTRL -> USU : insertar_usuario(id_personal, id_sucursal, id_cargo, modulo_id, username, usua_password)
USU --> CTRL : registro exitoso

CTRL -> BIT : registrarBitacora(id_usuario, accion_realizada, fecha_accion)
BIT --> CTRL : registro exitoso

CTRL -> GUIForm : mostrar confirmación
GUIForm -> GUILista : actualizar lista de usuarios

ADM -> GUILista : Selecciona usuario y clic en "Editar"
GUILista -> CTRL : obtenerUsuario(id_usuario)
CTRL -> USU : consultar_usuario(id_usuario)
USU --> CTRL : datos_usuario
CTRL -> GUIForm : mostrar formulario con datos cargados

ADM -> GUIForm : Modifica datos y clic en "Guardar"
GUIForm -> CTRL : editarUsuario(id_usuario, datos_usuario)

CTRL -> PER : verificar_personal(id_personal)
PER --> CTRL : datos_personal

CTRL -> SUC : verificar_sucursal(id_sucursal)
SUC --> CTRL : descripcion_sucursal, estado_sucursal

CTRL -> CAR : verificar_cargo(id_cargo)
CAR --> CTRL : cargo_descripcion, estado_cargo

CTRL -> MOD : verificar_modulo(modulo_id)
MOD --> CTRL : modulo_id, modulo_descri

CTRL -> USU : actualizar_usuario(id_usuario, datos_usuario)
USU --> CTRL : actualización exitosa

CTRL -> BIT : registrarBitacora(id_usuario, accion_realizada, fecha_accion)
BIT --> CTRL : registro exitoso

CTRL -> GUIForm : mostrar confirmación
GUIForm -> GUILista : actualizar lista de usuarios

ADM -> GUILista : Selecciona usuario y clic en "Eliminar"
GUILista -> ADM : Solicitar confirmación

ADM -> GUILista : Confirma eliminación
GUILista -> CTRL : eliminarUsuario(id_usuario)

CTRL -> USU : validar_vinculos_activos(id_usuario)
USU --> CTRL : sin vínculos activos / con vínculos activos

CTRL -> USU : eliminar_usuario(id_usuario)
USU --> CTRL : eliminación exitosa

CTRL -> BIT : registrarBitacora(id_usuario, accion_realizada, fecha_accion)
BIT --> CTRL : registro exitoso

CTRL -> GUILista : mostrar confirmación o error
GUILista -> ADM : actualizar lista principal

@enduml
