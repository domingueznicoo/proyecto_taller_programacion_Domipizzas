@startuml
title Registrar Orden de Compra – Botón: Guardar / Editar / Anular

autonumber

actor "Encargado de Compras" as ENC
boundary "GUI Lista de Órdenes de Compra" as GUILista
boundary "GUI Orden de Compra" as GUIForm
control "OrdenCompraController" as CTRL
control "conexion" as CONX
entity "orden_de_compra" as OC
entity "orden_detalle_compra" as OCDET
entity "presupuesto_compra" as PRESU
entity "presupuesto_detalle_compra" as PRESUDET
entity "proveedor" as PROV
entity "materia_prima" as MP
entity "tipo_iva" as TIVA
entity "usuarios" as USU
entity "sucursales" as SUC
entity "bitacora" as BIT

ENC -> GUILista : Accede a Compras → Órdenes de Compra
GUILista -> CTRL : listarOrdenes()
CTRL -> CONX : conector()
CONX --> CTRL : Connection
CTRL -> OC : consultar_ordenes()
OC --> CTRL : lista_ordenes
CTRL -> GUILista : mostrar lista principal

ENC -> GUILista : Clic en "Nuevo"
GUILista -> GUIForm : abrir formulario de OC

GUIForm -> CTRL : cargarContexto()
CTRL -> USU : consultar_usuario(id_usuario)
USU --> CTRL : datos_usuario
CTRL -> SUC : consultar_sucursal(id_sucursal)
SUC --> CTRL : datos_sucursal
CTRL -> PROV : consultar_proveedores_activos()
PROV --> CTRL : lista_proveedores
CTRL -> GUIForm : mostrar contexto y proveedores

ENC -> GUIForm : Selecciona proveedor
GUIForm -> CTRL : cargarPresupuestosEmitidos(id_proveedor)
CTRL -> PRESU : consultar_presupuestos_emitidos(id_proveedor)
PRESU --> CTRL : lista_presupuestos_emitidos
CTRL -> GUIForm : mostrar presupuestos emitidos

ENC -> GUIForm : Selecciona presupuesto
GUIForm -> CTRL : precargarDetallePresupuesto(id_presupuesto_compra)

CTRL -> PRESU : consultar_presupuesto(id_presupuesto_compra)
PRESU --> CTRL : datos_presupuesto

CTRL -> PRESUDET : consultar_detalle_presupuesto(id_presupuesto_compra)
PRESUDET --> CTRL : detalle_presupuesto

CTRL -> MP : consultar_materias_primas(detalle_presupuesto)
MP --> CTRL : datos_materias_primas

CTRL -> TIVA : consultar_tipo_iva(iva_id)
TIVA --> CTRL : datos_iva

CTRL -> GUIForm : mostrar detalle precargado

ENC -> GUIForm : Revisa condiciones de pago
GUIForm -> CTRL : validarDetalle(detalle)
CTRL --> GUIForm : detalle válido / inválido
GUIForm -> ENC : Mostrar error si existen datos inválidos

ENC -> GUIForm : Clic en "Guardar"
GUIForm -> CTRL : registrarOrden(datos_orden, detalle)

CTRL -> OC : insertar_orden(orden_fecha, orden_estado, orden_total, id_presupuesto_compra, id_proveedor, id_sucursal, id_usuario, orden_condicion)
OC --> CTRL : id_orden_compra

CTRL -> OCDET : insertar_detalle_orden(id_orden_compra, id_materia_prima, oc_cantidad_compra, oc_precio_compra)
OCDET --> CTRL : detalle registrado

CTRL -> PRESU : actualizar_estado_aprobado(id_presupuesto_compra)
PRESU --> CTRL : actualización exitosa

CTRL -> BIT : registrarBitacora(id_usuario, accion_realizada, fecha_accion)
BIT --> CTRL : registro exitoso

CTRL -> GUIForm : mostrar confirmación
GUIForm -> GUILista : actualizar lista principal

ENC -> GUILista : Selecciona OC Emitida y clic en "Editar"
GUILista -> CTRL : obtenerOrden(id_orden_compra)

CTRL -> OC : consultar_orden(id_orden_compra)
OC --> CTRL : datos_orden

CTRL -> OCDET : consultar_detalle_orden(id_orden_compra)
OCDET --> CTRL : detalle_orden

CTRL -> GUIForm : mostrar formulario con datos cargados

ENC -> GUIForm : Ajusta condiciones de pago
GUIForm -> CTRL : validarCondicionPago(orden_condicion)
CTRL --> GUIForm : condición válida / inválida
GUIForm -> ENC : Mostrar error si existen datos inválidos

ENC -> GUIForm : Clic en "Guardar"
GUIForm -> CTRL : editarOrden(id_orden_compra, orden_condicion)

CTRL -> OC : actualizar_orden(id_orden_compra, orden_condicion)
OC --> CTRL : actualización exitosa

CTRL -> BIT : registrarBitacora(id_usuario, accion_realizada, fecha_accion)
BIT --> CTRL : registro exitoso

CTRL -> GUIForm : mostrar confirmación
GUIForm -> GUILista : actualizar lista principal

ENC -> GUILista : Selecciona OC y clic en "Anular"
GUILista -> ENC : Solicitar confirmación

ENC -> GUILista : Confirma anulación
GUILista -> CTRL : anularOrden(id_orden_compra)

CTRL -> OC : verificar_estado_emitida(id_orden_compra)
OC --> CTRL : estado_orden

CTRL -> OC : validar_vinculos_posteriores(id_orden_compra)
OC --> CTRL : sin vínculos / con vínculos

CTRL -> OC : actualizar_estado_anulado(id_orden_compra)
OC --> CTRL : anulación exitosa

CTRL -> PRESU : actualizar_estado_emitido(id_presupuesto_compra)
PRESU --> CTRL : actualización exitosa

CTRL -> BIT : registrarBitacora(id_usuario, accion_realizada, fecha_accion)
BIT --> CTRL : registro exitoso

CTRL -> GUILista : mostrar confirmación o error
GUILista -> ENC : actualizar lista principal

@enduml
