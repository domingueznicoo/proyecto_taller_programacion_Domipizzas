@startuml
title Consultar y Generar Libro de Compras – Botón: Generar / Imprimir

autonumber

actor "Jefe de Compras" as JEFE
boundary "GUI Libro de Compras" as GUIForm
control "LibroCompraController" as CTRL
control "conexion" as CONX
entity "factura_compra" as FAC
entity "factura_detalle_compra" as FACDET
entity "nota_compra" as NOTA
entity "nota_detalle_compra" as NOTADET
entity "iva_compra" as IVAC
entity "tipo_iva" as TIVA
entity "proveedor" as PROV
entity "usuarios" as USU
entity "sucursales" as SUC
entity "bitacora" as BIT

JEFE -> GUIForm : Accede a Compras → Libro de Compras

GUIForm -> CTRL : validarSesion()
CTRL -> USU : consultar_usuario(id_usuario)
USU --> CTRL : datos_usuario
CTRL -> SUC : consultar_sucursal(id_sucursal)
SUC --> CTRL : datos_sucursal
CTRL -> GUIForm : mostrar formulario de filtros

JEFE -> GUIForm : Selecciona período y filtros
GUIForm -> CTRL : validarPeriodo(periodo_desde, periodo_hasta)
CTRL --> GUIForm : período válido / inválido
GUIForm -> JEFE : Mostrar error si el período es inválido

JEFE -> GUIForm : Clic en "Generar informe"
GUIForm -> CTRL : consultarDocumentos(periodo_desde, periodo_hasta, filtros)

CTRL -> CONX : conector()
CONX --> CTRL : Connection

CTRL -> FAC : consultar_facturas_vigentes(periodo, filtros)
FAC --> CTRL : lista_facturas

CTRL -> FACDET : consultar_detalle_facturas(lista_facturas)
FACDET --> CTRL : detalle_facturas

CTRL -> NOTA : consultar_notas_vigentes(periodo, filtros)
NOTA --> CTRL : lista_notas

CTRL -> NOTADET : consultar_detalle_notas(lista_notas)
NOTADET --> CTRL : detalle_notas

CTRL -> IVAC : consultar_iva_compras(periodo, filtros)
IVAC --> CTRL : iva_facturas

CTRL -> TIVA : consultar_tipos_iva()
TIVA --> CTRL : tipos_iva

CTRL -> PROV : consultar_proveedores_documentos()
PROV --> CTRL : datos_proveedores

CTRL -> CTRL : consolidarFacturas(lista_facturas, detalle_facturas, iva_facturas)
CTRL -> CTRL : consolidarNotas(lista_notas, detalle_notas)
CTRL -> CTRL : calcularTotalesIva()

CTRL -> GUIForm : validarResultados(documentos_consolidados)
GUIForm -> JEFE : Mostrar aviso si no existen resultados en el período

CTRL -> BIT : registrarBitacora(id_usuario, accion_realizada, fecha_accion)
BIT --> CTRL : registro exitoso

CTRL -> GUIForm : mostrar vista previa y resumen
GUIForm -> JEFE : Mostrar libro consolidado en pantalla

JEFE -> GUIForm : Clic en "Ver documento"
GUIForm -> CTRL : verDocumento(tipo_documento, id_documento)
CTRL -> FAC : consultar_factura_si_corresponde(id_documento)
FAC --> CTRL : datos_factura / no aplica
CTRL -> NOTA : consultar_nota_si_corresponde(id_documento)
NOTA --> CTRL : datos_nota / no aplica
CTRL -> GUIForm : mostrar detalle del documento

JEFE -> GUIForm : Clic en "Imprimir / PDF"
GUIForm -> CTRL : generarPDF(documentos_consolidados, filtros)
CTRL --> GUIForm : archivo PDF generado
GUIForm -> JEFE : Abrir PDF en nueva pestaña


@enduml
