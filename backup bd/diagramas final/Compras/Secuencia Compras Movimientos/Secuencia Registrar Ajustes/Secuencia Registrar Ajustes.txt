@startuml
title Registrar Ajustes – Botón: Guardar / Editar / Anular

autonumber

actor "Encargado de Compras" as ENC
boundary "GUI Lista de Ajustes" as GUILista
boundary "GUI Ajustes" as GUIForm
control "AjusteController" as CTRL
control "conexion" as CONX
entity "ajustes" as AJU
entity "ajustes_detalle" as AJUDET
entity "motivo" as MOT
entity "deposito" as DEP
entity "materia_prima" as MP
entity "stock_materia_prima" as STOCK
entity "usuarios" as USU
entity "bitacora" as BIT

ENC -> GUILista : Accede a Compras → Ajustes
GUILista -> CTRL : listarAjustes()
CTRL -> CONX : conector()
CONX --> CTRL : Connection
CTRL -> AJU : consultar_ajustes()
AJU --> CTRL : lista_ajustes
CTRL -> GUILista : mostrar lista principal

ENC -> GUILista : Clic en "Nuevo"
GUILista -> GUIForm : abrir formulario de ajuste

GUIForm -> CTRL : cargarContexto()
CTRL -> USU : consultar_usuario(id_usuario)
USU --> CTRL : datos_usuario
CTRL -> DEP : consultar_depositos()
DEP --> CTRL : lista_depositos
CTRL -> MOT : consultar_motivos_ajuste()
MOT --> CTRL : lista_motivos
CTRL -> MP : consultar_materias_primas_activas()
MP --> CTRL : lista_materias_primas
CTRL -> GUIForm : mostrar contexto, depósitos, motivos y materias primas

ENC -> GUIForm : Selecciona depósito, motivo, materia prima y cantidad
ENC -> GUIForm : Clic en "Agregar ítem"

GUIForm -> CTRL : validarItem(deposito_id, id_materia_prima, id_motivo, ajuste_cantidad)
CTRL --> GUIForm : ítem válido / inválido
GUIForm -> ENC : Mostrar error si existen datos inválidos

GUIForm -> CTRL : agregarItem(id_materia_prima, id_motivo, ajuste_cantidad)
CTRL -> STOCK : consultar_stock(id_materia_prima, deposito_id)
STOCK --> CTRL : id_stock, cantidad_existente
CTRL -> GUIForm : agregar ítem al detalle

ENC -> GUIForm : Clic en "Guardar"
GUIForm -> CTRL : validarDetalle(detalle)
CTRL --> GUIForm : detalle válido / inválido
GUIForm -> ENC : Mostrar error si el detalle está vacío

GUIForm -> CTRL : validarStockResultante(detalle, deposito_id)
CTRL -> STOCK : consultar_stock_actual(detalle, deposito_id)
STOCK --> CTRL : stock_actual
CTRL --> GUIForm : stock resultante válido / inválido
GUIForm -> ENC : Mostrar error si el stock resultante es negativo

GUIForm -> CTRL : registrarAjuste(datos_ajuste, detalle)

CTRL -> AJU : insertar_ajuste(ajuste_fecha, ajuste_estado, deposito_id, id_usuario)
AJU --> CTRL : id_ajuste

CTRL -> AJUDET : insertar_detalle(id_ajuste, id_materia_prima, id_stock, ajuste_cantidad, id_motivo)
AJUDET --> CTRL : detalle registrado

CTRL -> STOCK : actualizar_stock(id_stock, ajuste_cantidad)
STOCK --> CTRL : stock actualizado

CTRL -> BIT : registrarBitacora(id_usuario, accion_realizada, fecha_accion)
BIT --> CTRL : registro exitoso

CTRL -> GUIForm : mostrar confirmación
GUIForm -> GUILista : actualizar lista principal

ENC -> GUILista : Selecciona ajuste Emitido y clic en "Editar"
GUILista -> CTRL : obtenerAjuste(id_ajuste)

CTRL -> AJU : consultar_ajuste(id_ajuste)
AJU --> CTRL : datos_ajuste

CTRL -> AJUDET : consultar_detalle_ajuste(id_ajuste)
AJUDET --> CTRL : detalle_ajuste

CTRL -> GUIForm : mostrar formulario con datos cargados

ENC -> GUIForm : Modifica ítems o cantidades
GUIForm -> CTRL : validarDetalle(detalle)
CTRL --> GUIForm : detalle válido / inválido
GUIForm -> ENC : Mostrar error si existen datos inválidos

GUIForm -> CTRL : validarStockResultante(detalle, deposito_id)
CTRL -> STOCK : consultar_stock_actual(detalle, deposito_id)
STOCK --> CTRL : stock_actual
CTRL --> GUIForm : stock resultante válido / inválido

ENC -> GUIForm : Clic en "Guardar"
GUIForm -> CTRL : editarAjuste(id_ajuste, datos_ajuste, detalle)

CTRL -> AJU : actualizar_ajuste(id_ajuste, datos_ajuste)
AJU --> CTRL : actualización exitosa

CTRL -> AJUDET : actualizar_detalle_ajuste(id_ajuste, detalle)
AJUDET --> CTRL : detalle actualizado

CTRL -> STOCK : ajustar_stock_diferencia(id_ajuste, detalle)
STOCK --> CTRL : stock ajustado

CTRL -> BIT : registrarBitacora(id_usuario, accion_realizada, fecha_accion)
BIT --> CTRL : registro exitoso

CTRL -> GUIForm : mostrar confirmación
GUIForm -> GUILista : actualizar lista principal

ENC -> GUILista : Selecciona ajuste y clic en "Anular"
GUILista -> ENC : Solicitar confirmación

ENC -> GUILista : Confirma anulación
GUILista -> CTRL : anularAjuste(id_ajuste)

CTRL -> AJU : consultar_ajuste(id_ajuste)
AJU --> CTRL : datos_ajuste

CTRL -> AJUDET : consultar_detalle_ajuste(id_ajuste)
AJUDET --> CTRL : detalle_ajuste

CTRL -> STOCK : revertir_stock_ajuste(detalle_ajuste)
STOCK --> CTRL : stock revertido

CTRL -> AJU : actualizar_estado_anulado(id_ajuste)
AJU --> CTRL : anulación exitosa

CTRL -> BIT : registrarBitacora(id_usuario, accion_realizada, fecha_accion)
BIT --> CTRL : registro exitoso

CTRL -> GUILista : mostrar confirmación o error
GUILista -> ENC : actualizar lista principal

@enduml
