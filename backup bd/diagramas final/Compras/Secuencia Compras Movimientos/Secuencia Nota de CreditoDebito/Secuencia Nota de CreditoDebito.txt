@startuml
title Registrar Nota de Crédito / Débito – Botón: Guardar / Anular

autonumber

actor "Encargado de Compras" as ENC
boundary "GUI Lista de Notas" as GUILista
boundary "GUI Nota de Compra" as GUIForm
control "NotaCompraController" as CTRL
control "conexion" as CONX
entity "nota_compra" as NOTA
entity "nota_detalle_compra" as NOTADET
entity "motivo" as MOT
entity "factura_compra" as FAC
entity "factura_detalle_compra" as FACDET
entity "proveedor" as PROV
entity "materia_prima" as MP
entity "tipo_iva" as TIVA
entity "iva_compra" as IVAC
entity "cuentas_pagar" as CP
entity "stock_materia_prima" as STOCK
entity "usuarios" as USU
entity "sucursales" as SUC
entity "bitacora" as BIT

ENC -> GUILista : Accede a Compras → Notas
GUILista -> CTRL : listarNotas()
CTRL -> CONX : conector()
CONX --> CTRL : Connection
CTRL -> NOTA : consultar_notas()
NOTA --> CTRL : lista_notas
CTRL -> GUILista : mostrar lista principal

ENC -> GUILista : Clic en "Nuevo"
GUILista -> GUIForm : abrir formulario de nota

GUIForm -> CTRL : cargarContexto()
CTRL -> USU : consultar_usuario(id_usuario)
USU --> CTRL : datos_usuario
CTRL -> SUC : consultar_sucursal(id_sucursal)
SUC --> CTRL : datos_sucursal
CTRL -> PROV : consultar_proveedores_activos()
PROV --> CTRL : lista_proveedores
CTRL -> MOT : consultar_motivos()
MOT --> CTRL : lista_motivos
CTRL -> GUIForm : mostrar contexto, proveedores y motivos

ENC -> GUIForm : Selecciona tipo de nota
ENC -> GUIForm : Selecciona proveedor
GUIForm -> CTRL : cargarFacturasEmitidas(id_proveedor)

CTRL -> FAC : consultar_facturas_emitidas(id_proveedor)
FAC --> CTRL : lista_facturas_emitidas
CTRL -> GUIForm : mostrar facturas emitidas

ENC -> GUIForm : Selecciona factura
GUIForm -> CTRL : precargarDetalleFactura(id_factura_compra)

CTRL -> FAC : consultar_factura(id_factura_compra)
FAC --> CTRL : datos_factura

CTRL -> FACDET : consultar_detalle_factura(id_factura_compra)
FACDET --> CTRL : detalle_factura

CTRL -> MP : consultar_materias_primas(detalle_factura)
MP --> CTRL : datos_materias_primas

CTRL -> TIVA : consultar_tipo_iva(iva_id)
TIVA --> CTRL : datos_iva

CTRL -> GUIForm : mostrar detalle precargado

ENC -> GUIForm : Selecciona motivo, ajusta ítems/importes e ingresa timbrado y número
GUIForm -> CTRL : calcularTotales(detalle)
CTRL --> GUIForm : subtotal, IVA y total

GUIForm -> CTRL : validarNota(datos_nota, detalle)
CTRL --> GUIForm : datos válidos / inválidos
GUIForm -> ENC : Mostrar error si existen datos inválidos

ENC -> GUIForm : Clic en "Guardar"
GUIForm -> CTRL : registrarNota(datos_nota, detalle)

CTRL -> NOTA : insertar_nota(nota_compra_tipo, nota_compra_fecha, nota_nro, nota_compra_timbrado, nota_total, id_usuario, id_sucursal, id_proveedor, id_motivo, id_factura_compra)
NOTA --> CTRL : id_nota_compra

CTRL -> NOTADET : insertar_detalle_nota(id_nota_compra, id_materia_prima, nota_compra_cantidad, tipo_iva, nota_precio)
NOTADET --> CTRL : detalle registrado

CTRL -> CP : actualizar_cuenta_pagar(id_factura_compra, nota_compra_tipo, nota_total)
CP --> CTRL : cuenta actualizada

CTRL -> IVAC : actualizar_iva_compra(id_factura_compra)
IVAC --> CTRL : IVA actualizado

CTRL -> FAC : actualizar_total_factura(id_factura_compra, nota_compra_tipo, nota_total)
FAC --> CTRL : factura actualizada

CTRL -> STOCK : actualizar_stock_si_corresponde(id_materia_prima, cantidad)
STOCK --> CTRL : stock actualizado / no aplica

CTRL -> BIT : registrarBitacora(id_usuario, accion_realizada, fecha_accion)
BIT --> CTRL : registro exitoso

CTRL -> GUIForm : mostrar confirmación
GUIForm -> GUILista : actualizar lista principal

ENC -> GUILista : Selecciona nota y clic en "Anular"
GUILista -> ENC : Solicitar confirmación

ENC -> GUILista : Confirma anulación
GUILista -> CTRL : anularNota(id_nota_compra)

CTRL -> NOTA : consultar_nota(id_nota_compra)
NOTA --> CTRL : datos_nota

CTRL -> NOTA : actualizar_estado_anulada(id_nota_compra)
NOTA --> CTRL : anulación exitosa

CTRL -> CP : revertir_cuenta_pagar(id_factura_compra, nota_compra_tipo, nota_total)
CP --> CTRL : cuenta revertida

CTRL -> IVAC : revertir_iva_compra(id_factura_compra)
IVAC --> CTRL : IVA revertido

CTRL -> FAC : revertir_total_factura(id_factura_compra, nota_compra_tipo, nota_total)
FAC --> CTRL : factura revertida

CTRL -> STOCK : revertir_stock_si_corresponde(id_nota_compra)
STOCK --> CTRL : stock revertido / no aplica

CTRL -> BIT : registrarBitacora(id_usuario, accion_realizada, fecha_accion)
BIT --> CTRL : registro exitoso

CTRL -> GUILista : mostrar confirmación o error
GUILista -> ENC : actualizar lista principal

@enduml
