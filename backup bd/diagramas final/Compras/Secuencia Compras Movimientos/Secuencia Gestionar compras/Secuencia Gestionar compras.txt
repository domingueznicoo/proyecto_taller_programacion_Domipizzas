@startuml
title Gestionar Compras – Factura de Compra

autonumber

actor "Encargado de Compras" as ENC
boundary "GUI Lista de Facturas" as GUILista
boundary "GUI Factura de Compra" as GUIForm
control "CompraController" as CTRL
control "conexion" as CONX
entity "factura_compra" as FAC
entity "factura_detalle_compra" as FACDET
entity "orden_de_compra" as OC
entity "orden_detalle_compra" as OCDET
entity "proveedor" as PROV
entity "materia_prima" as MP
entity "tipo_iva" as TIVA
entity "iva_compra" as IVAC
entity "cuentas_pagar" as CP
entity "stock_materia_prima" as STOCK
entity "nota_remision_compra" as NRC
entity "usuarios" as USU
entity "sucursales" as SUC
entity "bitacora" as BIT

ENC -> GUILista : Accede a Compras → Gestionar Compras
GUILista -> CTRL : listarFacturas()
CTRL -> CONX : conector()
CONX --> CTRL : Connection
CTRL -> FAC : consultar_facturas()
FAC --> CTRL : lista_facturas
CTRL -> GUILista : mostrar lista principal

ENC -> GUILista : Clic en "Nueva Factura"
GUILista -> GUIForm : abrir formulario de factura

GUIForm -> CTRL : cargarContexto()
CTRL -> USU : consultar_usuario(id_usuario)
USU --> CTRL : datos_usuario
CTRL -> SUC : consultar_sucursal(id_sucursal)
SUC --> CTRL : datos_sucursal
CTRL -> PROV : consultar_proveedores_activos()
PROV --> CTRL : lista_proveedores
CTRL -> GUIForm : mostrar contexto y proveedores

ENC -> GUIForm : Selecciona proveedor
GUIForm -> CTRL : cargarOrdenesEmitidas(id_proveedor)
CTRL -> OC : consultar_ordenes_emitidas(id_proveedor)
OC --> CTRL : lista_ordenes_emitidas
CTRL -> GUIForm : mostrar órdenes emitidas

ENC -> GUIForm : Selecciona orden de compra
GUIForm -> CTRL : precargarDetalleOrden(id_orden_compra)

CTRL -> OC : consultar_orden(id_orden_compra)
OC --> CTRL : datos_orden

CTRL -> OCDET : consultar_detalle_orden(id_orden_compra)
OCDET --> CTRL : detalle_orden

CTRL -> MP : consultar_materias_primas(detalle_orden)
MP --> CTRL : datos_materias_primas

CTRL -> TIVA : consultar_tipo_iva(iva_id)
TIVA --> CTRL : datos_iva

CTRL -> GUIForm : mostrar detalle precargado

ENC -> GUIForm : Ingresa número factura, timbrado, fecha, cuotas e interés
ENC -> GUIForm : Selecciona nota de remisión si corresponde

GUIForm -> CTRL : validarFactura(datos_factura, detalle)
CTRL --> GUIForm : datos válidos / inválidos
GUIForm -> ENC : Mostrar error si existen datos inválidos

GUIForm -> CTRL : calcularTotales(detalle)
CTRL --> GUIForm : IVA y total calculados

ENC -> GUIForm : Clic en "Guardar"
GUIForm -> CTRL : registrarFactura(datos_factura, detalle)

CTRL -> FAC : insertar_factura(numero_factura, timbrado, fact_fecha_compra, fac_total, fac_estado, fac_plazo, id_usuario, id_sucursal, id_orden_compra, tipo_operacion)
FAC --> CTRL : id_factura_compra

CTRL -> FACDET : insertar_detalle_factura(id_factura_compra, id_materia_prima, fac_cantidad, fac_precio, fac_iva)
FACDET --> CTRL : detalle registrado

CTRL -> IVAC : insertar_iva_compra(id_factura_compra, iva_fecha, iva_exento, iva_5, iva_10)
IVAC --> CTRL : registro exitoso

CTRL -> CP : generar_cuenta_pagar(id_factura_compra, id_proveedor, monto_total, plazo_cuenta, nro_cuota)
CP --> CTRL : cuenta generada

CTRL -> OC : actualizar_estado_facturada(id_orden_compra)
OC --> CTRL : actualización exitosa

CTRL -> STOCK : actualizar_stock_si_corresponde(id_materia_prima, cantidad)
STOCK --> CTRL : stock actualizado / no aplica

CTRL -> BIT : registrarBitacora(id_usuario, accion_realizada, fecha_accion)
BIT --> CTRL : registro exitoso

CTRL -> GUIForm : mostrar confirmación
GUIForm -> GUILista : actualizar lista principal

ENC -> GUILista : Selecciona factura Pendiente y clic en "Aprobar"
GUILista -> ENC : Solicitar confirmación

ENC -> GUILista : Confirma aprobación
GUILista -> CTRL : aprobarFactura(id_factura_compra)

CTRL -> FAC : consultar_factura(id_factura_compra)
FAC --> CTRL : datos_factura

CTRL -> FACDET : consultar_detalle_factura(id_factura_compra)
FACDET --> CTRL : detalle_factura

CTRL -> IVAC : actualizar_iva_compra(id_factura_compra)
IVAC --> CTRL : actualización exitosa

CTRL -> CP : actualizar_cuenta_pagar(id_factura_compra, tipo_operacion)
CP --> CTRL : cuenta actualizada

CTRL -> FAC : actualizar_estado_emitida(id_factura_compra)
FAC --> CTRL : actualización exitosa

CTRL -> OC : actualizar_estado_finalizado(id_orden_compra)
OC --> CTRL : actualización exitosa

CTRL -> NRC : finalizar_nota_remision_si_corresponde(id_orden_compra)
NRC --> CTRL : finalizada / no aplica

CTRL -> STOCK : consolidar_stock_si_corresponde(id_materia_prima, cantidad)
STOCK --> CTRL : stock consolidado / no aplica

CTRL -> BIT : registrarBitacora(id_usuario, accion_realizada, fecha_accion)
BIT --> CTRL : registro exitoso

CTRL -> GUILista : mostrar confirmación
GUILista -> ENC : actualizar lista principal

ENC -> GUILista : Selecciona factura Pendiente y clic en "Anular"
GUILista -> ENC : Solicitar confirmación

ENC -> GUILista : Confirma anulación
GUILista -> CTRL : anularFactura(id_factura_compra)

CTRL -> FAC : verificar_estado_pendiente(id_factura_compra)
FAC --> CTRL : estado_factura

CTRL -> CP : validar_anulacion_cuenta(id_factura_compra)
CP --> CTRL : permite anulación / no permite

CTRL -> FAC : actualizar_estado_anulada(id_factura_compra)
FAC --> CTRL : anulación exitosa

CTRL -> CP : anular_cuenta_pagar(id_factura_compra)
CP --> CTRL : cuenta anulada

CTRL -> STOCK : revertir_stock_si_corresponde(id_factura_compra)
STOCK --> CTRL : stock revertido / no aplica

CTRL -> OC : actualizar_estado_emitida(id_orden_compra)
OC --> CTRL : actualización exitosa

CTRL -> BIT : registrarBitacora(id_usuario, accion_realizada, fecha_accion)
BIT --> CTRL : registro exitoso

CTRL -> GUILista : mostrar confirmación o error
GUILista -> ENC : actualizar lista principal

@enduml
