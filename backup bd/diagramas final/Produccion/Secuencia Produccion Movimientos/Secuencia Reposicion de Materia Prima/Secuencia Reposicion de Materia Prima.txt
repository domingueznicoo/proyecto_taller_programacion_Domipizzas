@startuml
title Registrar Reposición de Materia Prima – Botón: Guardar / Editar / Anular

autonumber

actor "Encargado de Producción" as ENC
boundary "GUI Lista Reposición de MP" as GUILista
boundary "GUI Reposición de MP" as GUIForm
control "ReposicionMateriaController" as CTRL
control "conexion" as CONX
entity "reposicion_materia" as REP
entity "reposicion_materia_detalle" as REPDET
entity "pedido_materia_produccion" as PEDMP
entity "pedido_materia_detalle_produccion" as PEDDET
entity "materia_prima" as MP
entity "stock_materia_prima" as STOCK
entity "deposito" as DEP
entity "usuarios" as USU
entity "bitacora" as BIT

ENC -> GUILista : Accede a Producción → Reposición de MP
GUILista -> CTRL : listarReposiciones()
CTRL -> CONX : conector()
CONX --> CTRL : Connection
CTRL -> REP : consultar_reposiciones()
REP --> CTRL : lista_reposiciones
CTRL -> GUILista : mostrar lista principal

ENC -> GUILista : Clic en "Nueva reposición"
GUILista -> GUIForm : abrir formulario de reposición

GUIForm -> CTRL : cargarContexto()
CTRL -> USU : consultar_usuario(id_usuario)
USU --> CTRL : datos_usuario
CTRL -> PEDMP : consultar_pedidos_pendientes_parciales()
PEDMP --> CTRL : lista_pedidos
CTRL -> GUIForm : mostrar contexto y pedidos disponibles

ENC -> GUIForm : Selecciona pedido de materia prima
GUIForm -> CTRL : cargarDetallePedido(id_pedido_materia)

CTRL -> PEDMP : consultar_pedido_materia(id_pedido_materia)
PEDMP --> CTRL : datos_pedido

CTRL -> DEP : consultar_deposito(deposito_id)
DEP --> CTRL : datos_deposito

CTRL -> PEDDET : consultar_detalle_pedido(id_pedido_materia)
PEDDET --> CTRL : detalle_pedido

CTRL -> MP : consultar_materias_primas(detalle_pedido)
MP --> CTRL : datos_materias_primas

CTRL -> GUIForm : mostrar depósito y detalle con pendientes

ENC -> GUIForm : Ingresa cantidad a reponer por ítem
GUIForm -> CTRL : validarItem(id_materia_prima, cantidad_reponer, pendiente)
CTRL --> GUIForm : ítem válido / inválido
GUIForm -> ENC : Mostrar error si cantidad excede pendiente o es inválida

ENC -> GUIForm : Clic en "Registrar reposición"
GUIForm -> CTRL : validarDetalle(detalle)
CTRL --> GUIForm : detalle válido / inválido
GUIForm -> ENC : Mostrar error si no hay ítems con cantidad

GUIForm -> CTRL : registrarReposicion(datos_reposicion, detalle)

CTRL -> REP : insertar_reposicion(reposicion_fecha, reposicion_estado, id_pedido_materia, deposito_id, id_usuario)
REP --> CTRL : id_reposicion_materia

CTRL -> REPDET : insertar_detalle_reposicion(id_reposicion_materia, id_materia_prima, cantidad_repuesta)
REPDET --> CTRL : detalle registrado

CTRL -> STOCK : incrementar_stock_materia(id_materia_prima, deposito_id, cantidad_repuesta)
STOCK --> CTRL : stock actualizado

CTRL -> PEDDET : actualizar_cantidad_repuesta(id_pedido_materia, id_materia_prima, cantidad_repuesta)
PEDDET --> CTRL : detalle pedido actualizado

CTRL -> PEDMP : recalcular_estado_pedido(id_pedido_materia)
PEDMP --> CTRL : estado actualizado

CTRL -> BIT : registrarBitacora(id_usuario, accion_realizada, fecha_accion)
BIT --> CTRL : registro exitoso

CTRL -> GUIForm : mostrar confirmación
GUIForm -> GUILista : actualizar lista principal

ENC -> GUILista : Selecciona reposición Registrada y clic en "Editar"
GUILista -> CTRL : obtenerReposicion(id_reposicion_materia)

CTRL -> REP : consultar_reposicion(id_reposicion_materia)
REP --> CTRL : datos_reposicion

CTRL -> REPDET : consultar_detalle_reposicion(id_reposicion_materia)
REPDET --> CTRL : detalle_reposicion

CTRL -> GUIForm : mostrar formulario con datos cargados

ENC -> GUIForm : Ajusta fecha o cantidades
GUIForm -> CTRL : editarReposicion(id_reposicion_materia, datos_reposicion, detalle)

CTRL -> STOCK : validarStockReversion(id_reposicion_materia)
STOCK --> CTRL : permite reversión / no permite

CTRL -> STOCK : revertir_stock_anterior(id_reposicion_materia)
STOCK --> CTRL : stock revertido

CTRL -> PEDDET : revertir_cantidad_repuesta_anterior(id_reposicion_materia)
PEDDET --> CTRL : cantidad repuesta revertida

CTRL -> REP : actualizar_reposicion(id_reposicion_materia, datos_reposicion)
REP --> CTRL : actualización exitosa

CTRL -> REPDET : actualizar_detalle_reposicion(id_reposicion_materia, detalle)
REPDET --> CTRL : detalle actualizado

CTRL -> STOCK : aplicar_nuevo_stock(id_materia_prima, deposito_id, cantidad_repuesta)
STOCK --> CTRL : stock actualizado

CTRL -> PEDDET : aplicar_nueva_cantidad_repuesta(id_pedido_materia, id_materia_prima, cantidad_repuesta)
PEDDET --> CTRL : cantidad repuesta actualizada

CTRL -> PEDMP : recalcular_estado_pedido(id_pedido_materia)
PEDMP --> CTRL : estado actualizado

CTRL -> BIT : registrarBitacora(id_usuario, accion_realizada, fecha_accion)
BIT --> CTRL : registro exitoso

CTRL -> GUIForm : mostrar confirmación
GUIForm -> GUILista : actualizar lista principal

ENC -> GUILista : Selecciona reposición y clic en "Anular"
GUILista -> ENC : Solicitar confirmación

ENC -> GUILista : Confirma anulación
GUILista -> CTRL : anularReposicion(id_reposicion_materia)

CTRL -> REP : consultar_reposicion(id_reposicion_materia)
REP --> CTRL : datos_reposicion

CTRL -> REPDET : consultar_detalle_reposicion(id_reposicion_materia)
REPDET --> CTRL : detalle_reposicion

CTRL -> STOCK : validarStockReversion(id_reposicion_materia)
STOCK --> CTRL : permite reversión / no permite

CTRL -> STOCK : disminuir_stock_materia(id_materia_prima, deposito_id, cantidad_repuesta)
STOCK --> CTRL : stock actualizado

CTRL -> PEDDET : revertir_cantidad_repuesta(id_pedido_materia, id_materia_prima, cantidad_repuesta)
PEDDET --> CTRL : cantidad repuesta revertida

CTRL -> PEDMP : recalcular_estado_pedido(id_pedido_materia)
PEDMP --> CTRL : estado actualizado

CTRL -> REP : actualizar_estado_anulado(id_reposicion_materia)
REP --> CTRL : anulación exitosa

CTRL -> BIT : registrarBitacora(id_usuario, accion_realizada, fecha_accion)
BIT --> CTRL : registro exitoso

CTRL -> GUILista : mostrar confirmación o error
GUILista -> ENC : actualizar lista principal

@enduml
