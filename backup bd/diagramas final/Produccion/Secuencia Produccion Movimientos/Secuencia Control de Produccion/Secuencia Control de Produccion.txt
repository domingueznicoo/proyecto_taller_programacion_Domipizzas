@startuml
title Registrar Control de Producción – Botón: Guardar / Editar / Anular

autonumber

actor "Encargado de Producción" as ENC
boundary "GUI Lista Control de Producción" as GUILista
boundary "GUI Control de Producción" as GUIForm
control "ControlProduccionController" as CTRL
control "conexion" as CONX
entity "control_produccion" as CPROD
entity "control_produccion_detalle" as CDET
entity "control_produccion_consumo" as CCONS
entity "orden_produccion" as OP
entity "orden_detalle_produccion" as OPDET
entity "etapa_produccion" as ETAPA
entity "etapa_detalle_produccion" as EDET
entity "materia_prima" as MP
entity "stock_materia_prima" as STOCK
entity "productos" as PROD
entity "inspectores" as INSP
entity "usuarios" as USU
entity "sucursales" as SUC
entity "bitacora" as BIT

ENC -> GUILista : Accede a Producción → Control de producción
GUILista -> CTRL : listarControles()
CTRL -> CONX : conector()
CONX --> CTRL : Connection
CTRL -> CPROD : consultar_controles()
CPROD --> CTRL : lista_controles
CTRL -> GUILista : mostrar lista principal

ENC -> GUILista : Clic en "Nuevo control"
GUILista -> GUIForm : abrir formulario de control

GUIForm -> CTRL : cargarContexto()
CTRL -> USU : consultar_usuario(id_usuario)
USU --> CTRL : datos_usuario
CTRL -> SUC : consultar_sucursal(id_sucursal)
SUC --> CTRL : datos_sucursal
CTRL -> OP : consultar_ordenes_pendientes_en_proceso()
OP --> CTRL : lista_ordenes
CTRL -> INSP : consultar_inspectores_activos()
INSP --> CTRL : lista_inspectores
CTRL -> GUIForm : mostrar contexto, OP e inspectores

ENC -> GUIForm : Selecciona OP
GUIForm -> CTRL : cargarProductosOrden(orden_id)

CTRL -> OPDET : consultar_detalle_orden(orden_id)
OPDET --> CTRL : productos_pendientes

CTRL -> PROD : consultar_productos(productos_pendientes)
PROD --> CTRL : datos_productos

CTRL -> GUIForm : mostrar productos de la OP

ENC -> GUIForm : Selecciona producto
GUIForm -> CTRL : determinarSiguienteEtapa(orden_id, producto_id)

CTRL -> ETAPA : consultar_ruta_producto(producto_id)
ETAPA --> CTRL : datos_ruta

CTRL -> EDET : consultar_siguiente_etapa(etapa_id, producto_id)
EDET --> CTRL : datos_etapa

CTRL -> GUIForm : mostrar siguiente etapa

ENC -> GUIForm : Ingresa cantidad procesada, inspector y observación
ENC -> GUIForm : Selecciona materia prima e ingresa cantidad consumida
ENC -> GUIForm : Clic en "Agregar consumo"

GUIForm -> CTRL : validarConsumo(id_materia_prima, cantidad_consumida)
CTRL -> STOCK : consultar_stock(id_materia_prima)
STOCK --> CTRL : cantidad_existente
CTRL --> GUIForm : consumo válido / inválido
GUIForm -> ENC : Mostrar error si stock insuficiente o cantidad inválida

GUIForm -> CTRL : agregarConsumo(id_materia_prima, cantidad_consumida)
CTRL -> GUIForm : agregar consumo al detalle

ENC -> GUIForm : Clic en "Guardar"
GUIForm -> CTRL : validarControl(datos_control, detalle_consumos)
CTRL --> GUIForm : control válido / inválido
GUIForm -> ENC : Mostrar error si existen datos inválidos

GUIForm -> CTRL : registrarControl(datos_control, detalle_consumos)

CTRL -> CPROD : insertar_control(orden_id, control_fecha, control_estado, id_inspectores, id_usuario, producto_id, etapa_id, control_observacion)
CPROD --> CTRL : control_id

CTRL -> CDET : insertar_detalle_control(control_id, producto_id, control_cantidad, control_descri)
CDET --> CTRL : detalle registrado

CTRL -> CCONS : insertar_consumo(control_id, id_materia_prima, cantidad_consumida)
CCONS --> CTRL : consumo registrado

CTRL -> STOCK : descontar_stock(id_materia_prima, cantidad_consumida)
STOCK --> CTRL : stock actualizado

CTRL -> OPDET : actualizar_cantidad_pendiente(orden_id, producto_id, control_cantidad)
OPDET --> CTRL : pendiente actualizado

CTRL -> OP : actualizar_estado_orden(orden_id)
OP --> CTRL : estado actualizado

CTRL -> BIT : registrarBitacora(id_usuario, accion_realizada, fecha_accion)
BIT --> CTRL : registro exitoso

CTRL -> GUIForm : mostrar confirmación
GUIForm -> GUILista : actualizar lista principal

ENC -> GUILista : Selecciona control Registrado y clic en "Editar"
GUILista -> CTRL : obtenerControl(control_id)

CTRL -> CPROD : consultar_control(control_id)
CPROD --> CTRL : datos_control

CTRL -> CDET : consultar_detalle_control(control_id)
CDET --> CTRL : detalle_control

CTRL -> CCONS : consultar_consumos(control_id)
CCONS --> CTRL : consumos_control

CTRL -> GUIForm : mostrar formulario con datos cargados

ENC -> GUIForm : Modifica inspector u observación
GUIForm -> CTRL : editarControl(control_id, id_inspectores, control_observacion)

CTRL -> INSP : validar_inspector_activo(id_inspectores)
INSP --> CTRL : inspector válido / inválido

CTRL -> CPROD : actualizar_control(control_id, id_inspectores, control_observacion)
CPROD --> CTRL : actualización exitosa

CTRL -> BIT : registrarBitacora(id_usuario, accion_realizada, fecha_accion)
BIT --> CTRL : registro exitoso

CTRL -> GUIForm : mostrar confirmación
GUIForm -> GUILista : actualizar lista principal

ENC -> GUILista : Selecciona control Registrado y clic en "Anular"
GUILista -> ENC : Solicitar confirmación

ENC -> GUILista : Confirma anulación
GUILista -> CTRL : anularControl(control_id)

CTRL -> CPROD : verificar_estado_registrado(control_id)
CPROD --> CTRL : estado_control

CTRL -> CCONS : consultar_consumos(control_id)
CCONS --> CTRL : consumos_control

CTRL -> STOCK : reponer_stock(id_materia_prima, cantidad_consumida)
STOCK --> CTRL : stock repuesto

CTRL -> OPDET : restaurar_cantidad_pendiente(orden_id, producto_id, control_cantidad)
OPDET --> CTRL : pendiente restaurado

CTRL -> CPROD : actualizar_estado_anulado(control_id)
CPROD --> CTRL : anulación exitosa

CTRL -> OP : actualizar_estado_orden(orden_id)
OP --> CTRL : estado actualizado

CTRL -> BIT : registrarBitacora(id_usuario, accion_realizada, fecha_accion)
BIT --> CTRL : registro exitoso

CTRL -> GUILista : mostrar confirmación o error
GUILista -> ENC : actualizar lista principal

@enduml
